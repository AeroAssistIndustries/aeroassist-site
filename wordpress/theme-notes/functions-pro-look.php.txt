 *              'live' = every visitor; 'off' = never.

if ( ! defined( 'AA_PRO_MODE' ) ) {
	define( 'AA_PRO_MODE', 'preview' );
}
function aa_pro_look_on() {
	if ( 'live' === AA_PRO_MODE ) {
		return true;
	}
	if ( 'off' === AA_PRO_MODE ) {
		return false;
	}
	return current_user_can( 'edit_theme_options' ) || ( isset( $_GET['look'] ) && 'pro' === $_GET['look'] );
}
add_filter( 'body_class', function ( $classes ) {
	if ( aa_pro_look_on() ) {
		$classes[] = 'aa-pro';
	}
	return $classes;
} );
